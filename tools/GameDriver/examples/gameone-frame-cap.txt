# GameOne: the frame cap. In an empty world's free play, turn "Show timings" on in the settings and
# save the readings uncapped (as fast as it can draw: "Timings fps" well over the screen's rate),
# then turn "Cap frames at the screen's rate" on and save them again (the screen's rate, 60 on a
# 60 Hz screen, "waiting for the GPU" taking up the rest of each frame), then turn both off again.
# Run with GAMEONE_WORLDS set to a new, empty folder, so the world made has its preset's name (a
# new world's is made unique among those saved) and the driver's worlds stay out of yours. The
# world list's buttons are clicked by their markers, wherever they are.
# GAMEONE_WORLDS=<empty folder> dotnet run tools/GameDriver/drive.cs -- src/GameOne/bin/Debug/net10.0/GameOne.exe tools/GameDriver/examples/gameone-frame-cap.txt --out shots
0    until Menu = Worlds
1    click marker "New: Empty world"
0    until Menu = Missions
1    click marker "Mission: FREE PLAY"
0    until Menu = None
1    tap escape          # the game menu ...
0    until Menu = Game
0.5  move 0 71
1    click 0 71          # ... Settings
0    until Menu = Settings
0.5  move -160 -16
1    click -160 -16       # Show timings: on
1.5  tap escape          # back to the game menu ...
2.5  tap escape          # ... and to flying
0    until Menu = None
4    shot uncapped       # a few seconds in, for a second's average or two
4    readings uncapped
4.5  tap escape
0    until Menu = Game
0.5  move 0 71
1    click 0 71          # Settings
0    until Menu = Settings
0.5  move -160 43
1    click -160 43       # Cap frames at the screen's rate: on
1.5  shot settings-capped
2    tap escape
3    tap escape
0    until Menu = None
4    shot capped
4    readings capped
4.5  tap escape
0    until Menu = Game
0.5  move 0 71
1    click 0 71          # Settings
0    until Menu = Settings
0.5  move -160 43
1    click -160 43       # Cap: off again
1.5  move -160 -16
2    click -160 -16       # Show timings: off
2.5  tap escape
3.5  tap escape
0    until Menu = None
2    quit
